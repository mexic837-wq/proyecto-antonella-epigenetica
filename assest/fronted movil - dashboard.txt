                        const radius = circle.r.baseVal.value;
                        const circumference = radius * 2 * Math.PI;
                        circle.style.strokeDasharray = `${circumference} ${circumference}`;
                        // 45% completion
                        const offset = circumference - (0.45 * circumference);
                        setTimeout(() => {
                            circle.style.strokeDashoffset = offset;
                        }, 100);
                    }
                }, 300);

                // Resultados
                const resLoad = document.getElementById('resultadosLoading');
                if(resLoad) {
                    resLoad.style.opacity = '0';
                    resLoad.style.transition = 'opacity 0.3s ease';
                    setTimeout(() => resLoad.style.display = 'none', 300);
                }
            }, 1200);

            // Nav Drawer Logic
            const menuBtn = document.getElementById('menuBtn');
            const navDrawer = document.getElementById('navDrawer');
            const navBackdrop = document.getElementById('navDrawerBackdrop');

            const toggleDrawer = () => {
                const isOpen = !navDrawer.classList.contains('-translate-x-full');
                if(isOpen) {
                    navDrawer.classList.add('-translate-x-full');
                    navBackdrop.classList.remove('open');
                } else {
                    navDrawer.classList.remove('-translate-x-full');
                    navBackdrop.classList.add('open');
                }
            };

            if(menuBtn) menuBtn.addEventListener('click', toggleDrawer);
            if(navBackdrop) navBackdrop.addEventListener('click', toggleDrawer);

            // Chat Bottom Sheet Logic
            const chatFab = document.getElementById('chatFab');
            const chatSheet = document.getElementById('chatSheet');
            const chatBackdrop = document.getElementById('chatBackdrop');
            const closeChatBtn = document.getElementById('closeChatBtn');
            const chatHandle = document.getElementById('chatHandle');

            const toggleChat = () => {
                chatSheet.classList.toggle('open');
                chatBackdrop.classList.toggle('open');
            };

            if(chatFab) chatFab.addEventListener('click', toggleChat);
            if(closeChatBtn) closeChatBtn.addEventListener('click', toggleChat);
            if(chatBackdrop) chatBackdrop.addEventListener('click', toggleChat);
            if(chatHandle) {
                let startY = 0;
                chatHandle.addEventListener('touchstart', (e) => {
                    startY = e.touches[0].clientY;
                }, {passive: true});
                chatHandle.addEventListener('touchmove', (e) => {
                    const currentY = e.touches[0].clientY;
                    if (currentY - startY > 50) {
                        toggleChat();
                    }
                }, {passive: true});
                chatHandle.addEventListener('click', toggleChat);
            }
        });
    </script>
